cmake_minimum_required(VERSION 3.25)

if(NOT DEFINED CMAKE_C_COMPILER AND NOT DEFINED ENV{CC})
    find_program(LOM_CLANG clang REQUIRED)
    set(CMAKE_C_COMPILER "${LOM_CLANG}" CACHE FILEPATH "C compiler")
endif()
if(NOT DEFINED CMAKE_CXX_COMPILER AND NOT DEFINED ENV{CXX})
    find_program(LOM_CLANGXX clang++ REQUIRED)
    set(CMAKE_CXX_COMPILER "${LOM_CLANGXX}" CACHE FILEPATH "C++ compiler")
endif()
project(lom_native LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/linkmaps")
if(APPLE)
    add_compile_definitions(_DARWIN_C_SOURCE)
    add_compile_definitions(_FORTIFY_SOURCE=0)
    add_compile_options(-fno-common)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_link_options("LINKER:-map,${CMAKE_BINARY_DIR}/linkmaps/$<TARGET_PROPERTY:NAME>.map")
    set(LOM_LINK_GC_SECTIONS "LINKER:-dead_strip")
    set(CMAKE_C_LINKER_LAUNCHER "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build/macho_wrap.py")
    set(CMAKE_CXX_LINKER_LAUNCHER ${CMAKE_C_LINKER_LAUNCHER})
else()
    add_link_options("LINKER:-Map=${CMAKE_BINARY_DIR}/linkmaps/$<TARGET_PROPERTY:NAME>.map")
    set(LOM_LINK_GC_SECTIONS "LINKER:--gc-sections")
endif()
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "The build needs Clang. Configure a fresh build directory with -DCMAKE_C_COMPILER=clang -DCMAKE_CXX_COMPILER=clang++.")
endif()
add_compile_definitions(LOM_NATIVE_POINTERS=1 M2CTX=1)
add_compile_options(-ftrivial-auto-var-init=zero)
add_compile_options($<$<COMPILE_LANGUAGE:C>:-fms-extensions>
    $<$<COMPILE_LANGUAGE:C>:-Wno-deprecated-non-prototype>
    $<$<COMPILE_LANGUAGE:C>:-fwrapv>
    $<$<COMPILE_LANGUAGE:C>:-fno-strict-aliasing>
    $<$<COMPILE_LANGUAGE:C>:-fno-delete-null-pointer-checks>)
if(WIN32)
    add_compile_options($<$<COMPILE_LANGUAGE:C>:-mno-ms-bitfields>)
endif()
set(LOM_SANITIZE "" CACHE STRING "Value for -fsanitize= on every target (empty: none)")
option(LOM_SANITIZE_RECOVER "Continue after sanitizer reports instead of aborting" OFF)
if(LOM_SANITIZE)
    add_compile_options(-fsanitize=${LOM_SANITIZE} -fno-omit-frame-pointer
        "-fsanitize-ignorelist=${CMAKE_CURRENT_SOURCE_DIR}/cmake/ubsan_ignorelist.txt")
    if(NOT LOM_SANITIZE_RECOVER)
        add_compile_options(-fno-sanitize-recover=${LOM_SANITIZE})
    endif()
    add_link_options(-fsanitize=${LOM_SANITIZE})
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(LOM_PSYZ_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/psyz" CACHE PATH
    "Path to the PSY-Z source tree")

set(LOM_DECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/game")
include(cmake/native/build.cmake)
